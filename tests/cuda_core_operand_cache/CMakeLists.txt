cmake_minimum_required(VERSION 3.24)
project(cuda_core_operand_cache LANGUAGES CXX)
include(CTest)
add_executable(cuda-core-operand-cache ../cuda_core_operand_cache_tests.cpp)
target_include_directories(cuda-core-operand-cache PRIVATE ../../include)
target_compile_features(cuda-core-operand-cache PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(cuda-core-operand-cache PRIVATE /W4 /WX /utf-8)
else()
 target_compile_options(cuda-core-operand-cache PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME cuda-core-operand-cache COMMAND cuda-core-operand-cache)
