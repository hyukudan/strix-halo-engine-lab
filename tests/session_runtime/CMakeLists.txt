cmake_minimum_required(VERSION 3.24)
project(strix_session_runtime_host LANGUAGES CXX)
include(CTest)
option(STRIX_LEGACY_MAINS "Build legacy Gemma4 and MiniMax serve executables" ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(Threads REQUIRED)
if(MSVC)
    add_compile_options(/utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
add_library(strix_model_description STATIC "${REPO_ROOT}/src/model_description.cpp")
target_include_directories(strix_model_description PUBLIC "${REPO_ROOT}/include")
target_compile_features(strix_model_description PUBLIC cxx_std_20)
add_library(strix_model_registry STATIC "${REPO_ROOT}/src/model_registry.cpp")
target_link_libraries(strix_model_registry PUBLIC strix_model_description)
set(serve_sources json json_constraint constraint_selection bpe_tokenizer prompt_stream chat_template
    deepseek_chat glm_chat http_server engine api deployment request_fields health media)
add_library(strix_serve STATIC)
set(STRIX_BUILD_JSON "{\"format\":\"strix-build\",\"revision\":1,\"status\":\"unavailable\"}")
configure_file("${REPO_ROOT}/include/strix/build_info.hpp.in"
    "${CMAKE_CURRENT_BINARY_DIR}/generated/strix/build_info.hpp" @ONLY)
target_sources(strix_serve PRIVATE "${REPO_ROOT}/src/preflight.cpp" "${REPO_ROOT}/src/linux_uma_observation.cpp"
    "${REPO_ROOT}/src/uma_owner_inventory.cpp")
target_include_directories(strix_serve PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
foreach(source IN LISTS serve_sources)
    target_sources(strix_serve PRIVATE "${REPO_ROOT}/src/serve/${source}.cpp")
endforeach()
target_include_directories(strix_serve PUBLIC "${REPO_ROOT}/include")
target_include_directories(strix_serve PRIVATE "${REPO_ROOT}/src")
target_compile_features(strix_serve PUBLIC cxx_std_20)
target_link_libraries(strix_serve PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(strix_serve PUBLIC ws2_32)
endif()
# Use the same library and guarded contract tests as the root/minimal build.
include("${REPO_ROOT}/cmake/StrixSession.cmake")
strix_session_add_runtime()
if(MSVC)
    foreach(target strix_serve strix_model_registry strix_model_description)
        target_compile_options(${target} PRIVATE /EHsc /utf-8 /WX)
    endforeach()
    target_compile_options(strix_session_owner PRIVATE /WX)
    if(TARGET session_runtime_tests)
        target_compile_options(session_runtime_tests PRIVATE /WX)
    endif()
endif()

# The merged API owns the shared power receipt and media tables, including in host-only tests.
target_sources(strix_serve PRIVATE "${REPO_ROOT}/src/power/sampler.cpp")
target_link_libraries(strix_serve PUBLIC strix_session)
if(BUILD_TESTING)
    add_executable(serve_api_vision_tests "${REPO_ROOT}/tests/serve_api_vision_tests.cpp")
    target_link_libraries(serve_api_vision_tests PRIVATE strix_session_shell)
    strix_session_warnings(serve_api_vision_tests)
    if(MSVC)
        target_compile_options(serve_api_vision_tests PRIVATE /WX)
    endif()
    add_test(NAME serve_api_vision COMMAND serve_api_vision_tests)
    set_tests_properties(serve_api_vision PROPERTIES TIMEOUT 60)
endif()
