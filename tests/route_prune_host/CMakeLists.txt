cmake_minimum_required(VERSION 3.24)
project(strix_route_prune_host LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_executable(route-prune-host ../route_prune_host_tests.cpp)
target_include_directories(route-prune-host PRIVATE ../../include ../../apps)
target_compile_features(route-prune-host PRIVATE cxx_std_20)
target_link_libraries(route-prune-host PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(route-prune-host PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(route-prune-host PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(route-prune-host PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME route-prune-parser COMMAND route-prune-host parser)
add_test(NAME route-prune-census COMMAND route-prune-host census)
add_test(NAME route-prune-controls-json COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/../route_prune_controls_test.py" "$<TARGET_FILE:route-prune-host>")
