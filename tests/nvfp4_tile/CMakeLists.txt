cmake_minimum_required(VERSION 3.24)
project(strix_nvfp4_tile LANGUAGES CXX)
enable_testing()
# Host oracle of the FAST4 tile GEMM staging (strix/cuda/core/nvfp4_tile.hpp); -DSTRIX_NVFP4_TILE_CUDA=ON adds the sm_120 device
# fixture + operator bench (GPU execution only through an operator ticket).
option(STRIX_NVFP4_TILE_CUDA "Compile the sm_120 device fixture and bench" OFF)
get_filename_component(STRIX_ROOT ../.. ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
add_library(strix_nvfp4_tile_ref STATIC ${STRIX_ROOT}/src/cuda/core/nvfp4_reference.cpp)
target_include_directories(strix_nvfp4_tile_ref PUBLIC ${STRIX_ROOT}/include)
add_executable(nvfp4_tile_host tile_host.cpp)
target_link_libraries(nvfp4_tile_host PRIVATE strix_nvfp4_tile_ref)
add_test(NAME nvfp4_tile_host COMMAND nvfp4_tile_host)
if(MSVC)
  target_compile_options(nvfp4_tile_host PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(nvfp4_tile_host PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(STRIX_NVFP4_TILE_CUDA)
  enable_language(CUDA)
  add_executable(nvfp4_tile_device tile_device.cu ${STRIX_ROOT}/src/cuda/core/nvfp4_tile.cu
               ${STRIX_ROOT}/src/cuda_core/norms/norms.cu ${STRIX_ROOT}/src/cuda/core/recur_chunk.cu)
  target_link_libraries(nvfp4_tile_device PRIVATE strix_nvfp4_tile_ref)
  set_target_properties(nvfp4_tile_device PROPERTIES CUDA_ARCHITECTURES 120a-real CUDA_STANDARD 20)
  target_compile_options(nvfp4_tile_device PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>)
  if(MSVC)
    target_compile_options(nvfp4_tile_device PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/EHsc,/Zc:preprocessor,/utf-8>)
  endif()
  add_test(NAME nvfp4_tile_device COMMAND nvfp4_tile_device)
  set_tests_properties(nvfp4_tile_device PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=0" TIMEOUT 110 LABELS "gpu;cuda")
endif()
