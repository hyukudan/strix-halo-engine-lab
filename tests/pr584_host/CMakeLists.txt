cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(strix_pr584_host LANGUAGES CXX)
    enable_testing()
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
    "${ROOT}/src/hip_deepseek41_prefill.hip"
    "${ROOT}/src/hip_ds41_layer.hip"
    "${ROOT}/src/core/f16_dense_q8a8.hpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
    "${ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
function(host_target name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE "${ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_features(${name} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
        target_compile_options(${name} PRIVATE /W4 /WX /utf-8 /fp:strict)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    endif()
endfunction()
host_target(pr584-orchestration orchestration.cpp)
add_test(NAME pr584-peer-ring-boundary COMMAND pr584-orchestration ring)
add_test(NAME pr584-flash-transition COMMAND pr584-orchestration precision)
add_test(NAME pr584-k-part-cancellation-demonstration COMMAND pr584-orchestration cancellation)

    target_link_libraries(pr584-cache PRIVATE ds41_prefix_owner_fixture)

    include("${ROOT}/cmake/StrixStateCache.cmake")
    target_sources(strix_state_cache PRIVATE "${ROOT}/src/deepseek41_attention_state.cpp")
    if(MSVC)
        target_compile_definitions(strix_state_cache PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    host_target(pr584-cache cache.cpp "${ROOT}/src/deepseek41_prefix_cache.cpp")
    target_link_libraries(pr584-cache PRIVATE strix_state_cache)
endif()
add_test(NAME pr584-dense-cache-threshold COMMAND pr584-cache)
