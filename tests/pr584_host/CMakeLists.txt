cmake_minimum_required(VERSION 3.24)
project(strix_pr584_host LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(PR584_BEFORE_FIX "Exercise the regressions against the unmodified PR head" OFF)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
    "${ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" "${PR584_BEFORE_FIX}" COMMAND_ERROR_IS_FATAL ANY)
function(host_target name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE "${ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_features(${name} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
        target_compile_options(${name} PRIVATE /W4 /WX /utf-8 /fp:strict)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    endif()
endfunction()
host_target(pr584-orchestration orchestration.cpp)
add_test(NAME pr584-peer-ring-boundary COMMAND pr584-orchestration ring)
add_test(NAME pr584-flash-transition COMMAND pr584-orchestration precision)
add_test(NAME pr584-k-part-cancellation-demonstration COMMAND pr584-orchestration cancellation)

    target_compile_definitions(strix_state_cache PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

    set(CACHE_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/before_cache.cpp")
endif()
