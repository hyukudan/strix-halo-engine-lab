cmake_minimum_required(VERSION 3.20)
project(audit716_host LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(audit716-host eager_restart_tests.cpp)
target_include_directories(audit716-host PRIVATE ../../include)
target_compile_features(audit716-host PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(audit716-host PRIVATE /W4 /WX /permissive-)
    target_compile_definitions(audit716-host PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(audit716-host PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME audit716-restart-failure COMMAND audit716-host failure)
add_test(NAME audit716-explicit-hip COMMAND audit716-host explicit)
add_test(NAME audit716-kfd-reports COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/kfdevict_tests.py")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    foreach(case deleted replaced denied)
        add_test(NAME audit716-restart-${case} COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/linux_restart_test.py" "$<TARGET_FILE:audit716-host>"
            "${CMAKE_CURRENT_BINARY_DIR}" "${case}")
    endforeach()
endif()
