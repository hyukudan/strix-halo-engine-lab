cmake_minimum_required(VERSION 3.24)
project(audit710_host LANGUAGES CXX)
enable_testing()
add_executable(qwen_cohort_context host.cpp ../../src/mtp_context_draft.cpp)
target_include_directories(qwen_cohort_context PRIVATE ../../include)
target_compile_features(qwen_cohort_context PRIVATE cxx_std_20)
if(MSVC)
    target_compile_definitions(qwen_cohort_context PRIVATE _CRT_SECURE_NO_WARNINGS)
    # Retain the inline cohort's references even in a Release host-only link test.
    target_compile_options(qwen_cohort_context PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /Od)
else()
    target_compile_options(qwen_cohort_context PRIVATE -Wall -Wextra -Werror -O0)
endif()
foreach(case IN ITEMS zero_depth copy rejection mixed tail shadow)
    add_test(NAME cohort_${case} COMMAND qwen_cohort_context ${case})
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME cohort_link_and_receipts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/contracts.py)
