cmake_minimum_required(VERSION 3.20)
project(strix_pr576_host LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(STRIX_ROOT "../.." ABSOLUTE)
find_package(Threads REQUIRED)
add_executable(direct-row-fetch-tests direct_row_fetch_tests.cpp)
target_include_directories(direct-row-fetch-tests PRIVATE "${STRIX_ROOT}/src")
target_link_libraries(direct-row-fetch-tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_definitions(direct-row-fetch-tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(direct-row-fetch-tests PRIVATE /W4 /WX)
else()
    target_compile_options(direct-row-fetch-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME throwing-sink-drain-reuse COMMAND direct-row-fetch-tests sink)
add_test(NAME entry-failure-ownership COMMAND direct-row-fetch-tests entry)
add_test(NAME ring-recycling-and-read-failure COMMAND direct-row-fetch-tests lifecycle)
set_tests_properties(throwing-sink-drain-reuse entry-failure-ownership ring-recycling-and-read-failure PROPERTIES TIMEOUT 20)

add_executable(engram-policy-tests engram_policy_tests.cpp "${STRIX_ROOT}/src/ds41_engram.cpp")
target_include_directories(engram-policy-tests PRIVATE "${STRIX_ROOT}/include" "${STRIX_ROOT}/src")
target_link_libraries(engram-policy-tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_definitions(engram-policy-tests PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    target_compile_options(engram-policy-tests PRIVATE /W4 /WX)
else()
    target_compile_options(engram-policy-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME engram-policy-and-fallback COMMAND engram-policy-tests)
set_tests_properties(engram-policy-and-fallback PROPERTIES TIMEOUT 20)

add_executable(engram-cache-tests "${STRIX_ROOT}/tests/ds41_engram_cache_tests.cpp"
    "${STRIX_ROOT}/src/ds41_engram.cpp")
target_include_directories(engram-cache-tests PRIVATE "${STRIX_ROOT}/include" "${STRIX_ROOT}/src")
target_link_libraries(engram-cache-tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_definitions(engram-cache-tests PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    target_compile_options(engram-cache-tests PRIVATE /W4 /WX)
else()
    target_compile_options(engram-cache-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME engram-cache-backends COMMAND engram-cache-tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    option(STRIX_REQUIRE_RING "Require native ring/direct coverage in the qualifying job" OFF)
    if(STRIX_REQUIRE_RING)
        add_test(NAME engram-cache-required-ring COMMAND engram-cache-tests --require-ring)
    endif()
else()
    add_test(NAME unavailable-ring-is-rejected COMMAND engram-cache-tests --require-ring)
    set_tests_properties(unavailable-ring-is-rejected PROPERTIES WILL_FAIL TRUE)
endif()
