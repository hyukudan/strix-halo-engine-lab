cmake_minimum_required(VERSION 3.20)
project(minimax_route_host LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(STRIX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ENGINE_SOURCE "${STRIX_ROOT}/src/minimax/engine/engine.hip" CACHE FILEPATH "Engine source (override for negative controls)")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
    "${STRIX_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" "${ENGINE_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(minimax-route-host regression.cpp)
add_executable(chunk-order "${STRIX_ROOT}/tests/chunk_order_contract_tests.cpp")
add_library(minimax-config-defaults STATIC "${STRIX_ROOT}/src/minimax/engine/mm_defaults.cpp")
add_executable(minimax-rank-config startup_config.cpp)
add_executable(minimax-serve-config startup_config.cpp)
target_link_libraries(minimax-rank-config PRIVATE minimax-config-defaults)
target_link_libraries(minimax-serve-config PRIVATE minimax-config-defaults)
target_compile_definitions(minimax-serve-config PRIVATE STRIX_TEST_MINIMAX_SERVE=1)
foreach(target minimax-route-host chunk-order minimax-config-defaults minimax-rank-config minimax-serve-config)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${STRIX_ROOT}/src" "${STRIX_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
target_compile_definitions(minimax-config-defaults PRIVATE STRIX_MM_GU4_SCALAR=1)
foreach(case startup resources decode routes mlp grouped-plan)
    add_test(NAME minimax-route-${case} COMMAND minimax-route-host ${case})
endforeach()
add_test(NAME chunk-order COMMAND chunk-order)
add_test(NAME minimax-prefill-order-startup COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/startup.py"
    "$<TARGET_FILE:minimax-rank-config>" "$<TARGET_FILE:minimax-serve-config>")
