cmake_minimum_required(VERSION 3.20)
project(strix_pr581_host LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(STRIX_ROOT "../.." ABSOLUTE)
set(STRIX_PR581_BASELINE "" CACHE STRING "Optional original PR revision for negative regression controls")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
    "${STRIX_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" "${STRIX_PR581_BASELINE}" COMMAND_ERROR_IS_FATAL ANY)
set(versions fixed)
if(STRIX_PR581_BASELINE)
    list(APPEND versions original)
endif()
foreach(version IN LISTS versions)
    foreach(suite aio ready)
        add_executable(${suite}-${version} "${CMAKE_CURRENT_BINARY_DIR}/${suite}-${version}.cpp")
        target_compile_features(${suite}-${version} PRIVATE cxx_std_20)
        if(MSVC)
            target_compile_options(${suite}-${version} PRIVATE /W4 /WX /permissive- /utf-8)
            target_compile_definitions(${suite}-${version} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${suite}-${version} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endforeach()
endforeach()
foreach(case partial-submit wait-error pressure-empty pressure-partial pressure-drained zero-submit interrupted
             invalid-zero invalid-range thread-failure)
    add_test(NAME aio-${case} COMMAND aio-fixed ${case})
    if(STRIX_PR581_BASELINE)
        add_test(NAME original-fails-${case} COMMAND aio-original ${case})
        set_tests_properties(original-fails-${case} PROPERTIES WILL_FAIL TRUE)
    endif()
endforeach()
foreach(case multi-window short-read)
    add_test(NAME aio-${case} COMMAND aio-fixed ${case})
endforeach()
foreach(case tail-route digest-boundary digest-slot)
    add_test(NAME ${case} COMMAND ready-fixed ${case})
    if(STRIX_PR581_BASELINE)
        add_test(NAME original-fails-${case} COMMAND ready-original ${case})
        set_tests_properties(original-fails-${case} PROPERTIES WILL_FAIL TRUE)
    endif()
endforeach()
add_test(NAME ready-row-coverage COMMAND ready-fixed row-coverage)
set_tests_properties(aio-pressure-empty aio-pressure-partial aio-pressure-drained PROPERTIES TIMEOUT 10)
