cmake_minimum_required(VERSION 3.24)
project(strix_nvfp4 LANGUAGES CXX)
enable_testing()
option(STRIX_NVFP4_CUDA "Compile native SM120 operator and device tests" OFF)
get_filename_component(STRIX_ROOT ../.. ABSOLUTE)
if(STRIX_NVFP4_CUDA)
  enable_language(CUDA)
endif()
add_subdirectory(${STRIX_ROOT}/src/cuda/core ${CMAKE_CURRENT_BINARY_DIR}/core)
add_executable(nvfp4_host host.cpp)
target_link_libraries(nvfp4_host PRIVATE strix_nvfp4_reference)
add_test(NAME nvfp4_host COMMAND nvfp4_host)
if(STRIX_NVFP4_CUDA)
  add_executable(nvfp4_device device.cu)
  target_link_libraries(nvfp4_device PRIVATE strix_nvfp4)
  find_library(STRIX_NVFP4_NVML_LIBRARY NAMES nvml nvidia-ml
    HINTS "${CMAKE_CUDA_COMPILER_TOOLKIT_ROOT}/lib/x64")
  if(STRIX_NVFP4_NVML_LIBRARY)
    target_link_libraries(nvfp4_device PRIVATE ${STRIX_NVFP4_NVML_LIBRARY})
    target_compile_definitions(nvfp4_device PRIVATE STRIX_NVFP4_NVML=1)
  endif()
  set_target_properties(nvfp4_device PROPERTIES CUDA_ARCHITECTURES 120a CUDA_STANDARD 17)
  add_test(NAME nvfp4_device COMMAND nvfp4_device)
  set_tests_properties(nvfp4_device PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=0" TIMEOUT 110)
endif()
