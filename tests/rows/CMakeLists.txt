cmake_minimum_required(VERSION 3.24)
project(strix_rows_tests LANGUAGES CXX)
enable_testing()

set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(strix-rows-plan-tests rows_plan_tests.cpp)
add_executable(strix-decode-projection-group-tests ../decode_projection_group_tests.cpp)
foreach(test strix-rows-plan-tests strix-decode-projection-group-tests)
    target_include_directories(${test} PRIVATE "${root}/include")
    target_compile_features(${test} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${test} PRIVATE /W4 /WX)
    else()
        target_compile_options(${test} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()

option(STRIX_ROWS_CUDA_TEST "Build the CUDA device bitwise fixture" OFF)
if(STRIX_ROWS_CUDA_TEST)
    enable_language(CUDA)
    add_executable(strix-cuda-rows-grouped-test rows_grouped_test.cu "${root}/src/cuda_core/dense/dense_mma.cu")
    set_target_properties(strix-cuda-rows-grouped-test PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
    target_include_directories(strix-cuda-rows-grouped-test PRIVATE "${root}/include" "${root}/src")
    target_compile_options(strix-cuda-rows-grouped-test PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false)
    if(MSVC)
        target_compile_definitions(strix-cuda-rows-grouped-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
        target_compile_options(strix-cuda-rows-grouped-test PRIVATE -Xcompiler=/EHsc,/Zc:preprocessor,/utf-8)
    endif()
endif()
