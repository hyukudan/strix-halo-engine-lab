cmake_minimum_required(VERSION 3.20)
project(qwen35_host_guards LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare.py" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(qwen35-guard-host host_tests.cpp)
target_include_directories(qwen35-guard-host PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(qwen35-guard-admission "${ROOT}/tests/qwen35_config_tests.cpp" "${ROOT}/src/qwen35/config.cpp" "${ROOT}/src/qwen35/family.cpp" "${ROOT}/src/gguf_reader.cpp" "${ROOT}/src/model_family_registry.cpp")
target_include_directories(qwen35-guard-admission PRIVATE "${ROOT}/include")
foreach(t qwen35-guard-host qwen35-guard-admission)
    target_compile_features(${t} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /WX /EHsc /utf-8)
        target_compile_definitions(${t} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME ${t} COMMAND ${t})
endforeach()
add_test(NAME qwen35-guard-runner COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/runner_test.py" "${CMAKE_CURRENT_BINARY_DIR}/runner")
add_test(NAME qwen35-guard-cmake-duplicates COMMAND "${CMAKE_COMMAND}" -S "${CMAKE_CURRENT_LIST_DIR}/cmake_duplicates" -B "${CMAKE_CURRENT_BINARY_DIR}/duplicates" -G "${CMAKE_GENERATOR}" "-DROOT=${ROOT}")
