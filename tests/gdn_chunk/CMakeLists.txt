cmake_minimum_required(VERSION 3.24)
project(strix_gdn_chunk LANGUAGES CXX)
get_filename_component(STRIX_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../.. ABSOLUTE)
# Host oracle of the FAST chunked GDN prefill (strix/cuda_core/gdn_chunk.hpp); -DSTRIX_GDN_CHUNK_CUDA=ON adds the sm_120 device
# fixture/bench (chunk kernels vs the serial gdn_prefill_layer, GPU ticket only).
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(gdn_chunk_host gdn_chunk_host.cpp)
target_include_directories(gdn_chunk_host PRIVATE ${STRIX_ROOT}/include)
add_test(NAME gdn_chunk_host COMMAND gdn_chunk_host)
if(MSVC)
  target_compile_options(gdn_chunk_host PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(gdn_chunk_host PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
enable_testing()
if(STRIX_GDN_CHUNK_CUDA)
  enable_language(CUDA)
  add_executable(gdn_chunk_device gdn_chunk_device.cu ${STRIX_ROOT}/src/cuda_core/gdn/gdn.cu ${STRIX_ROOT}/src/cuda_core/gdn/gdn_chunk.cu)
  target_include_directories(gdn_chunk_device PRIVATE ${STRIX_ROOT}/include)
  set_target_properties(gdn_chunk_device PROPERTIES CUDA_ARCHITECTURES 120a-real CUDA_STANDARD 20)
  target_compile_options(gdn_chunk_device PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>)
  if(MSVC)
    target_compile_options(gdn_chunk_device PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/EHsc,/Zc:preprocessor,/utf-8>)
  endif()
  add_test(NAME gdn_chunk_device COMMAND gdn_chunk_device)
  set_tests_properties(gdn_chunk_device PROPERTIES ENVIRONMENT "CUDA_VISIBLE_DEVICES=0" TIMEOUT 110 LABELS "gpu;cuda")
endif()
