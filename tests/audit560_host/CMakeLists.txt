cmake_minimum_required(VERSION 3.24)
project(strix_audit560_host LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
get_filename_component(ROOT ../.. ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ROOT}/apps/hip_mtp_decode_bridge.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py")
option(AUDIT560_ORIGINAL_CONTROL "Also require each regression to fail on unmodified HEAD" OFF)
set(kinds fixed)
if(AUDIT560_ORIGINAL_CONTROL)
    list(APPEND kinds original)
endif()
foreach(kind IN LISTS kinds)
    set(extra)
    if(kind STREQUAL original)
        set(extra --original)
    endif()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
        --root "${ROOT}" --out "${CMAKE_CURRENT_BINARY_DIR}/${kind}" ${extra}
        COMMAND_ERROR_IS_FATAL ANY)
    add_executable(audit560-${kind} adapter_tests.cpp "${ROOT}/src/serve/step_planner.cpp")
    target_include_directories(audit560-${kind} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${kind}" "${ROOT}/include")
    target_compile_features(audit560-${kind} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(audit560-${kind} PRIVATE /W4 /WX)
    else()
        target_compile_options(audit560-${kind} PRIVATE -Wall -Wextra -Werror -Wshadow -Wconversion)
    endif()
    foreach(finding IN ITEMS zero_draft policy_cohorts prefix_profiles prompt_accounting)
        add_test(NAME audit560-${kind}-${finding} COMMAND audit560-${kind} ${finding})
        if(kind STREQUAL original)
            set_tests_properties(audit560-${kind}-${finding} PROPERTIES WILL_FAIL TRUE)
        endif()
    endforeach()
endforeach()
