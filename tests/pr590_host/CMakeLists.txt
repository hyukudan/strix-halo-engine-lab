cmake_minimum_required(VERSION 3.24)
project(strix_pr590_host LANGUAGES CXX)
include(CTest)
set(STRIX_LEGACY_MAINS OFF CACHE BOOL "" FORCE)
add_subdirectory(../session_runtime session_runtime)
get_filename_component(STRIX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(strix_glm_session_owner STATIC "${STRIX_ROOT}/src/glm/serve/glm_session_owner.cpp")
target_include_directories(strix_glm_session_owner PUBLIC "${STRIX_ROOT}/src")
target_link_libraries(strix_glm_session_owner PUBLIC strix_session_owner)
if(MSVC)
    set(GLM_OWNER_MSVC_COMPAT "${CMAKE_CURRENT_LIST_DIR}/msvc_math.hpp" CACHE FILEPATH "" FORCE)
endif()
add_subdirectory(../glm_session_owner glm_owner)
target_sources(glm_owner_existing_cpu_tests PRIVATE "${STRIX_ROOT}/src/serve/conversation_keep.cpp"
    "${STRIX_ROOT}/src/serve/prefix_directory.cpp")
foreach(target strix_glm_session_owner glm_owner_native_fixture glm_owner_existing_cpu_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /WX)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Werror)
    endif()
endforeach()
add_test(NAME pr590-queued-successor COMMAND glm_owner_existing_cpu_tests --interleave-only)
set_tests_properties(pr590-queued-successor PROPERTIES TIMEOUT 30)
