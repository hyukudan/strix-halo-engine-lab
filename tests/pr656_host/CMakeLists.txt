cmake_minimum_required(VERSION 3.24)
project(strix_pr656_host LANGUAGES CXX)
include(CTest)
get_filename_component(STRIX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Reuse the existing host-only GLM fixture and its MSVC math compatibility shim.
add_subdirectory(../pr590_host pr590)
include("${STRIX_ROOT}/cmake/StrixPairState.cmake")
foreach(target strix-pair-state-owners-tests pair_state_serialized_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
add_test(NAME pr656-glm-import COMMAND glm_owner_existing_cpu_tests --audit656-import)
set_tests_properties(pr656-glm-import PROPERTIES TIMEOUT 120)
