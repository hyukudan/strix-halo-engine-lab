cmake_minimum_required(VERSION 3.24)
project(strix_pair_layercut_host LANGUAGES CXX)
include(CTest)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(strix-pair-layercut-tests "${root}/tests/pair_layercut_tests.cpp")
target_include_directories(strix-pair-layercut-tests PRIVATE "${root}/include")
target_compile_features(strix-pair-layercut-tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(strix-pair-layercut-tests PRIVATE /W4 /WX /EHsc /wd4996)
else()
    target_compile_options(strix-pair-layercut-tests PRIVATE -Wall -Wextra -Werror -Wshadow -Wsign-conversion)
endif()
add_test(NAME strix-pair-layercut-tests COMMAND strix-pair-layercut-tests)

add_executable(strix-pair-layercut-pack-tests "${root}/tests/pair_layercut_pack_tests.cpp"
    "${root}/src/minimax/rtxpf/cuda/mm_rtx_pack.cpp")
target_include_directories(strix-pair-layercut-pack-tests PRIVATE "${root}/src")
target_compile_features(strix-pair-layercut-pack-tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(strix-pair-layercut-pack-tests PRIVATE /W4 /WX /EHsc /wd4996)
endif()
add_test(NAME strix-pair-layercut-pack-tests COMMAND strix-pair-layercut-pack-tests "${CMAKE_CURRENT_BINARY_DIR}/pack-fixture")
