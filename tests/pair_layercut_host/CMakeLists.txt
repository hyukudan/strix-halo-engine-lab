cmake_minimum_required(VERSION 3.24)
project(strix_pair_layercut_host LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(strix-pair-layercut-tests "${root}/tests/pair_layercut_tests.cpp")
target_include_directories(strix-pair-layercut-tests PRIVATE "${root}/include")
target_compile_features(strix-pair-layercut-tests PRIVATE cxx_std_20)
target_link_libraries(strix-pair-layercut-tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(strix-pair-layercut-tests PRIVATE /W4 /WX /EHsc /wd4996)
else()
    target_compile_options(strix-pair-layercut-tests PRIVATE -Wall -Wextra -Werror -Wshadow -Wsign-conversion)
endif()
add_test(NAME strix-pair-layercut-tests COMMAND strix-pair-layercut-tests)

add_executable(strix-pair-layercut-pack-tests "${root}/tests/pair_layercut_pack_tests.cpp"
    "${root}/src/cuda/models/minimax/mm_rtx_pack.cpp")
target_include_directories(strix-pair-layercut-pack-tests PRIVATE "${root}/src")
target_compile_features(strix-pair-layercut-pack-tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(strix-pair-layercut-pack-tests PRIVATE /W4 /WX /EHsc /wd4996)
endif()
add_test(NAME strix-pair-layercut-pack-tests COMMAND strix-pair-layercut-pack-tests "${CMAKE_CURRENT_BINARY_DIR}/pack-fixture")

add_executable(strix-handoff-lanes-tests "${root}/tests/handoff_lanes_tests.cpp" "${root}/src/handoff.cpp" "${root}/src/bp_codec.cpp")
target_include_directories(strix-handoff-lanes-tests PRIVATE "${root}/include" "${root}/src")
target_compile_features(strix-handoff-lanes-tests PRIVATE cxx_std_20)
target_link_libraries(strix-handoff-lanes-tests PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(strix-handoff-lanes-tests PRIVATE ws2_32)
endif()
if(MSVC)
    target_compile_options(strix-handoff-lanes-tests PRIVATE /W4 /WX /EHsc /wd4996)
else()
    target_compile_options(strix-handoff-lanes-tests PRIVATE -Wall -Wextra -Werror -Wshadow -Wsign-conversion)
endif()
add_test(NAME strix-handoff-lanes-tests COMMAND strix-handoff-lanes-tests)
add_test(NAME strix-handoff-lanes-failed-write COMMAND strix-handoff-lanes-tests --failure)
add_test(NAME strix-handoff-lanes-rejection COMMAND strix-handoff-lanes-tests --rejection)
add_test(NAME strix-handoff-lanes-stopped COMMAND strix-handoff-lanes-tests --stopped)
add_test(NAME strix-handoff-lanes-mixed COMMAND strix-handoff-lanes-tests --mixed)
set_tests_properties(strix-handoff-lanes-tests strix-handoff-lanes-failed-write
    strix-handoff-lanes-rejection strix-handoff-lanes-stopped strix-handoff-lanes-mixed PROPERTIES TIMEOUT 25)

# Compile the actual rank adapter parser without pulling in the HIP execution functions.
file(READ "${root}/apps/minimax_layercut.inc" adapter)
string(FIND "${adapter}" "std::vector<std::string> cut_fields(" parser_start)
string(FIND "${adapter}" "std::vector<std::uint32_t> layercut_session(" parser_end)
math(EXPR parser_size "${parser_end} - ${parser_start}")
string(SUBSTRING "${adapter}" ${parser_start} ${parser_size} parser)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cut_arguments_under_test.inc" "${parser}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/apps/minimax_layercut.inc")
add_executable(strix-layercut-execution-parser-tests "${root}/tests/layercut_execution_parser_tests.cpp")
