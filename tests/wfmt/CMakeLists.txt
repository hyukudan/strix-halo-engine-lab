cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(strix_wfmt LANGUAGES CXX)
    enable_testing()
endif()
# Initialize BUILD_TESTING even when the root enters this directory before CTest.
include(CTest)
get_filename_component(WFMT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
set(STRIX_WFMT_BACKEND "host" CACHE STRING "Weight-format qualification backend")
option(STRIX_WFMT_CAPTURE "Enable diagnostic copies at existing host accumulation steps" OFF)
if(NOT COMMAND strix_target_warnings)
    function(strix_target_warnings target)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        endif()
    endfunction()
endif()
include(${WFMT_ROOT}/cmake/StrixWeight.cmake)
include(${WFMT_ROOT}/cmake/StrixBackendCpu.cmake)
include(${WFMT_ROOT}/cmake/StrixIntel.cmake)
add_executable(strix-wfmt-host-tests main.cpp contracts.cpp census.cpp compare.cpp
    reference.cpp geometry.cpp fixtures/iq3_s_cases.cpp
    resolve.cpp cache.cpp effective_config.cpp runtime.cpp readers.cpp admission.cpp reference_admission.cpp cpu.cpp intel.cpp
    ${WFMT_ROOT}/src/vision/gguf_weights.cpp
    ${WFMT_ROOT}/src/gemma4/model/safetensors.cpp
    ${WFMT_ROOT}/src/core/weight_archive.cpp)
target_link_libraries(strix-wfmt-host-tests PRIVATE strix-weight strix_backend_cpu strix_intel_core)
target_include_directories(strix-wfmt-host-tests PRIVATE ${WFMT_ROOT}/src ${WFMT_ROOT}/apps)
target_compile_definitions(strix-wfmt-host-tests PRIVATE STRIX_WFMT_SOURCE_DIR="${WFMT_ROOT}")
if(MSVC)
    target_compile_options(strix-weight PRIVATE /fp:strict)
    target_compile_options(strix-wfmt-host-tests PRIVATE /fp:strict /D_CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(strix-weight PRIVATE -ffp-contract=off)
    target_compile_options(strix-wfmt-host-tests PRIVATE -ffp-contract=off)
endif()
foreach(case contracts census compare reference geometry iq3s readers admission resolve cache effective-config runtime cpu intel strides arithmetic group-reads)
    add_test(NAME wfmt.${case} COMMAND strix-wfmt-host-tests ${case})
endforeach()
add_test(NAME wfmt.early-backend-registration COMMAND ${CMAKE_COMMAND}
    -DBINARY_ROOT=${CMAKE_CURRENT_BINARY_DIR} -DGENERATOR=${CMAKE_GENERATOR}
    -P ${CMAKE_CURRENT_LIST_DIR}/cmake_registration.cmake)
foreach(backend cpu intel)
    add_executable(strix-wfmt-${backend}-tests backend_main.cpp ${backend}.cpp compare.cpp)
    target_include_directories(strix-wfmt-${backend}-tests PRIVATE ${WFMT_ROOT}/src ${WFMT_ROOT}/include)
    if(backend STREQUAL "cpu")
        target_compile_definitions(strix-wfmt-${backend}-tests PRIVATE STRIX_WFMT_CASE_CPU=1)
        target_link_libraries(strix-wfmt-${backend}-tests PRIVATE strix_backend_cpu)
    else()
        target_link_libraries(strix-wfmt-${backend}-tests PRIVATE strix_intel_core)
    endif()
    if(MSVC)
        target_compile_options(strix-wfmt-${backend}-tests PRIVATE /fp:strict)
    else()
        target_compile_options(strix-wfmt-${backend}-tests PRIVATE -ffp-contract=off)
    endif()
endforeach()
add_test(NAME wfmt.cpu-binding COMMAND strix-wfmt-cpu-tests)
add_test(NAME wfmt.intel-binding COMMAND strix-wfmt-intel-tests)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    foreach(case formats native-fp8)
        string(REPLACE "-" "_" fixture ${case})
        add_executable(strix-ds41-${case}-tests ${WFMT_ROOT}/tests/ds41_${fixture}_tests.cpp)
        target_link_libraries(strix-ds41-${case}-tests PRIVATE strix-weight)
        add_test(NAME strix-ds41-${case}-tests COMMAND strix-ds41-${case}-tests)
    endforeach()
    add_executable(strix-ds41-trellis-identity-tests
        ${WFMT_ROOT}/tests/ds41_trellis_identity_tests.cpp ${WFMT_ROOT}/src/ds41_trellis.cpp)
    target_link_libraries(strix-ds41-trellis-identity-tests PRIVATE strix-weight)
    add_test(NAME strix-ds41-trellis-identity-tests COMMAND strix-ds41-trellis-identity-tests)
    foreach(case host tolerance_host)
        string(REPLACE "_host" "" source ${case})
        add_executable(nvfp4_${case} ${WFMT_ROOT}/tests/nvfp4/${source}.cpp)
        target_link_libraries(nvfp4_${case} PRIVATE strix-weight)
        add_test(NAME nvfp4_${case} COMMAND nvfp4_${case})
    endforeach()
endif()

if(UNIX)
    add_executable(strix-wfmt-legacy-reader-tests legacy_readers.cpp ${WFMT_ROOT}/src/glm/load/glm_rank_plan.cpp)
    target_include_directories(strix-wfmt-legacy-reader-tests PRIVATE ${WFMT_ROOT}/src)
    target_link_libraries(strix-wfmt-legacy-reader-tests PRIVATE strix-weight)
    add_test(NAME wfmt.glm-reader COMMAND strix-wfmt-legacy-reader-tests)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME wfmt.tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_tools.py)
add_test(NAME wfmt.legacy-source COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_legacy_compare.py)
set_tests_properties(wfmt.tools wfmt.legacy-source PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")

add_test(NAME wfmt.capture-source COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/capture_source.py)
set_tests_properties(wfmt.capture-source PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
