cmake_minimum_required(VERSION 3.24)
project(strix_dense_fixture_host LANGUAGES CXX)
enable_testing()
add_executable(strix-dense-fixture-checks-tests ../dense_fixture_checks_tests.cpp)
target_compile_features(strix-dense-fixture-checks-tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(strix-dense-fixture-checks-tests PRIVATE /W4 /WX)
else()
    target_compile_options(strix-dense-fixture-checks-tests PRIVATE -Wall -Wextra -Werror -Wshadow -Wsign-conversion)
endif()
add_test(NAME strix-dense-fixture-checks-tests COMMAND strix-dense-fixture-checks-tests)
# Compile the edited host translation unit too, without linking or building HIP.
add_library(vision-fixture-syntax OBJECT ../vision_encoder_hip_tests.cpp)
target_include_directories(vision-fixture-syntax PRIVATE ../../include)
target_compile_features(vision-fixture-syntax PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(vision-fixture-syntax PRIVATE /W4 /WX)
    target_compile_definitions(vision-fixture-syntax PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(vision-fixture-syntax PRIVATE -Wall -Wextra -Werror -Wshadow -Wsign-conversion)
endif()
