cmake_minimum_required(VERSION 3.24)
project(strix_inkling2_mini_host_tests LANGUAGES CXX)
enable_testing()
add_executable(inkling2-mini-schedule-test mini_schedule_test.cpp)
target_include_directories(inkling2-mini-schedule-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/inkling2")
target_compile_features(inkling2-mini-schedule-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(inkling2-mini-schedule-test PRIVATE /W4 /WX)
else()
  target_compile_options(inkling2-mini-schedule-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME inkling2-mini-schedule COMMAND inkling2-mini-schedule-test)
