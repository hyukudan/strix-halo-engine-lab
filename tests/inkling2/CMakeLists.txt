function(inkling2_mini_test name source)
    set(target "inkling2-mini-${name}-test")
    add_executable(${target} ${source})
    target_include_directories(${target} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/../../src"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(COMMAND strix_target_warnings)
        strix_target_warnings(${target})
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -ffp-contract=off)
    endif()
    add_test(NAME inkling2-mini-${name} COMMAND ${target})
    set_tests_properties(inkling2-mini-${name} PROPERTIES LABELS "cpu;inkling2")
endfunction()
