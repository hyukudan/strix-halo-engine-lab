cmake_minimum_required(VERSION 3.24)
project(strix_pr587_host LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(STRIX_ROOT "../.." ABSOLUTE)
add_executable(cycle_receipt_tests ../cycle_receipt_tests.cpp "${STRIX_ROOT}/src/serve/json.cpp")
add_executable(run_receipt_adapter_tests ../run_receipt_adapter_tests.cpp)
foreach(target cycle_receipt_tests run_receipt_adapter_tests)
    target_include_directories(${target} PRIVATE "${STRIX_ROOT}/include")
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
add_test(NAME cycle_receipt_tests COMMAND cycle_receipt_tests)
foreach(case stop restored rows prefill)
    add_test(NAME run_receipt_${case} COMMAND run_receipt_adapter_tests ${case})
    add_test(NAME run_receipt_${case}_wiring COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/source_contract.py" "${STRIX_ROOT}/apps/deepseek41_serve_main.cpp" ${case})
endforeach()
