cmake_minimum_required(VERSION 3.20)
project(strix_pr575_host LANGUAGES CXX)
enable_testing()
add_executable(strix-pr575-admission-tests ../pr575_admission_tests.cpp)
target_compile_features(strix-pr575-admission-tests PRIVATE cxx_std_20)
target_include_directories(strix-pr575-admission-tests PRIVATE ../../include)
if(MSVC)
    target_compile_definitions(strix-pr575-admission-tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(strix-pr575-admission-tests PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(strix-pr575-admission-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
foreach(case failure lazy pipeline)
    add_test(NAME pr575-${case} COMMAND strix-pr575-admission-tests ${case})
endforeach()
