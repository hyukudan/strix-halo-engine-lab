cmake_minimum_required(VERSION 3.24)
project(strix_gpu_queue LANGUAGES CXX HIP)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native DMA-BUF queue requires Linux")
endif()
set(STRIX_ENABLE_HIP_RUNTIME ON CACHE BOOL "Build native HIP runtime")
set(BUILD_TESTING OFF CACHE BOOL "Build explicit commissioning targets only")
get_filename_component(STRIX_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${STRIX_REPO}" "${CMAKE_CURRENT_BINARY_DIR}/engine" EXCLUDE_FROM_ALL)
if(NOT TARGET strix_hip_runtime)
    message(FATAL_ERROR "Actual HIP runtime/toolchain required")
endif()
add_library(strix_gpu_queue STATIC ../../src/linux_gpu_queue.cpp ../../src/gpu_queue_session.cpp)
target_compile_features(strix_gpu_queue PUBLIC cxx_std_20)
target_include_directories(strix_gpu_queue PUBLIC "${STRIX_REPO}/include")
target_compile_options(strix_gpu_queue PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
add_executable(strix-gpu-queue-visibility native_main.hip ../../src/hip_gpu_visibility_probe.hip ../direct_gpu_stream/bounded_tcp_control.cpp)
set_target_properties(strix-gpu-queue-visibility PROPERTIES HIP_STANDARD 20 HIP_STANDARD_REQUIRED ON HIP_ARCHITECTURES gfx1151)
target_compile_features(strix-gpu-queue-visibility PRIVATE cxx_std_20)
target_link_libraries(strix-gpu-queue-visibility PRIVATE strix_gpu_queue strix_hip_runtime)
target_compile_options(strix-gpu-queue-visibility PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing)
